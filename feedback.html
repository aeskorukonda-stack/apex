Submit Feedback / Testimonials (Requires Admin Approval) async function submitFeedback() { const name = document.getElementById("fbName").value.trim(); const role = document.getElementById("fbRole").value; const message = document.getElementById("fbMessage").value.trim(); const msgEl = document.getElementById("fbMsg"); if (!name || !message) { msgEl.textContent = "Please provide both your name and feedback."; msgEl.className = "msg error"; msgEl.classList.remove("hidden"); return; } // Set approved to false so it requires admin verification before appearing on ticker const { error } = await db.from("testimonials").insert([{ user_id: currentUser ? currentUser.id : null, name: name, role: role, message: message, approved: false }]); if (error) { msgEl.textContent = error.message; msgEl.className = "msg error"; } else { msgEl.textContent = "Thank you! Your feedback has been submitted for admin approval and will appear once verified."; msgEl.className = "msg success"; document.getElementById("fbMessage").value = ""; document.getElementById("fbName").value = ""; } msgEl.classList.remove("hidden"); } // 2. Submit Help / Query Request Directly to Admin Dashboard async function submitHelpQuery() { const name = document.getElementById("helpName").value.trim(); const contact = document.getElementById("helpContact").value.trim(); const category = document.getElementById("helpCategory").value; const details = document.getElementById("helpDetails").value.trim(); const msgEl = document.getElementById("helpMsg"); if (!name || !contact || !details) { msgEl.textContent = "Please complete all fields (name, contact, and query)."; msgEl.className = "msg error"; msgEl.classList.remove("hidden"); return; } // Insert ticket directly into Supabase support_tickets table (viewable in Admin Dashboard) const { error } = await db.from("support_tickets").insert([{ user_id: currentUser ? currentUser.id : null, name: name, contact: contact, category: category, message: details, status: 'Pending' }]); if (error) { msgEl.textContent = "Error sending message: " + error.message; msgEl.className = "msg error"; } else { msgEl.textContent = "Your request has been delivered directly to the admin team. We will review it shortly."; msgEl.className = "msg success"; document.getElementById("helpDetails").value = ""; } msgEl.classList.remove("hidden"); }
